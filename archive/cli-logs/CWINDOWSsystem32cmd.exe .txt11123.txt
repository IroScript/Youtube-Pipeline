    csv_file = export_dir / f'{tbl}.csv'
    with open(csv_file, 'w', newline='', encoding='utf-8-sig') as f:
        writer = csv.writer(f)
        writer.writerow(cols)
        writer.writerows(rows)

    results.append({
        'table': tbl,
        'rows': len(rows),
        'cols': len(cols),
        'path': str(csv_file)
    })
    print(f'Exported Table: {tbl:<25} | Rows: {len(rows):<5} | Cols: {len(cols):<3} | CSV: {csv_file}')

conn.close()
