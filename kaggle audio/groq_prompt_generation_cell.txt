import os
import json
import base64
from groq import Groq
from IPython.display import display, HTML

# 🔴 Groq API Key - Loaded from an external file (Do not hardcode!)
api_key_path = "/kaggle/working/groq_api_key.txt" 
# (You can also change this path to your Kaggle Input dataset path if needed)

try:
    with open(api_key_path, "r", encoding="utf-8") as f:
        GROQ_API_KEY = f.read().strip()
except FileNotFoundError:
    print(f"❌ Error: API key file not found at {api_key_path}.")
    print("Please create this file and paste your API key inside it, or update the path.")
    raise

client = Groq(api_key=GROQ_API_KEY)

# Find the latest transcription text file
import glob
audio_dir = "/kaggle/working/outputs/v10_combine/"
txt_files = glob.glob(os.path.join(audio_dir, "*_transcription.txt"))

if not txt_files:
    print("❌ No transcription files found in", audio_dir)
else:
    latest_txt = max(txt_files, key=os.path.getctime)
    print(f"✅ Found transcript: {latest_txt}")
    
    with open(latest_txt, "r", encoding="utf-8") as f:
        transcript_data = f.read()

    # 🔴 Character Gender Identity (Change to Female if needed)
    CHARACTER_GENDER = "Male"

    system_prompt = """You are an expert AI Video Director creating prompts for Veo 3.1.
You will receive an audio transcript with word-by-word timestamps. Your job is to group consecutive words into logical Scenes, and map out EXACTLY when actions should occur to sync perfectly with the spoken words.

CRITICAL RULES:
1. Target Durations allowed: ONLY 4, 6, or 8 seconds. DO NOT use any other duration.
2. Minimum Scene Duration & Granularity: Each generated scene must have a total audio duration of at least 3 seconds (so it fits Veo's 4s limit). HOWEVER, inside the 'full_combined_prompt', you MUST still break down this 4s slot into granular sub-second timeline actions matching the exact words (e.g. 0.0-0.5s, 0.5-1.2s, etc). Do not skip the micro-timing!
3. Veo duration must be at least 0.5s longer than the scene's audio duration.
4. Output strict JSON with a single root key "scenes" containing an array of objects.
5. TIMELINE PROMPTING (CRITICAL): Veo is not a strict timeline engine, but writing timeline ranges in the prompt vastly improves audio-video sync. Inside the "full_combined_prompt", you MUST use timestamp-based descriptive timelines (e.g. "0.0-1.2s: [action]. 1.2-3.0s: [action]"). Synchronize these visual descriptions directly with the exact word timestamps you receive.
6. CHARACTER IDENTITY: The main character voice is """ + CHARACTER_GENDER + """. Ensure all prompts describe the main character consistently as a """ + CHARACTER_GENDER + """.

JSON Structure for each object:
{
  "scene_number": 1,
  "start_time_seconds": 0.0,
  "end_time_seconds": 3.44,
  "audio_duration_seconds": 3.44,
  "veo_target_duration": 4,
  "aspect_ratio": "9:16",
  "priority": "lower",
  "openmontage_prompt": {
      "subject": "Detailed character action/expression mapped to the audio emotion",
      "environment": "Background and setting details",
      "camera": "Specific camera movement",
      "lighting": "Cinematic lighting",
      "consistency_notes": "Notes to keep character consistent"
  },
  "final_movement_direction": "Overall continuous camera motion for the entire scene (e.g., Constant slow push-in while tilting up)",
  "full_combined_prompt": "0.0-1.5s: A vast abandoned city under a blood-red sky. 1.5-2.8s: Dinosaurs emerge from the mist, their heavy footsteps echoing. 2.8-4.0s: Humanity watches from a hovering ship. OVERALL MOTION: Constant slow push-in while tilting up."
}
"""

    print("⏳ Sending request to Groq API... (Trying openai/gpt-oss-120b first)")
    json_output = ""
    
    # Try the big model first
    try:
        completion = client.chat.completions.create(
            model="openai/gpt-oss-120b",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": f"Transcript:\n{transcript_data}"}
            ],
            response_format={"type": "json_object"},
            temperature=0.3,
            max_tokens=4000
        )
        json_output = completion.choices[0].message.content
        print("✅ Success with openai/gpt-oss-120b!")
    except Exception as e:
        print(f"⚠️ Failed with openai/gpt-oss-120b ({e}). Falling back to llama-3.3-70b-versatile...")
        # Fallback to Llama 3.3
        completion = client.chat.completions.create(
            model="llama-3.3-70b-versatile",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": f"Transcript:\n{transcript_data}"}
            ],
            response_format={"type": "json_object"},
            temperature=0.3,
            max_tokens=4000
        )
        json_output = completion.choices[0].message.content
        print("✅ Success with llama-3.3-70b-versatile!")

    if json_output:
        # Save to JSON file
        json_path = latest_txt.replace("_transcription.txt", "_veo_prompts.json")
        with open(json_path, "w", encoding="utf-8") as f:
            f.write(json_output)
        
        print(f"📁 JSON saved to: {json_path}")
        
        # ⬇️ Auto-Download Script ⬇️
        with open(json_path, "rb") as f:
            b64 = base64.b64encode(f.read()).decode()
        
        filename = os.path.basename(json_path)
        html = f'''
        <a download="{filename}" id="download-json" href="data:application/json;base64,{b64}" target="_blank">
            <button style="padding: 10px 20px; font-size: 16px; background-color: #4CAF50; color: white; border: none; border-radius: 5px; cursor: pointer;">
                ⬇️ Download Veo Prompts JSON
            </button>
        </a>
        <script>
            document.getElementById('download-json').click();
        </script>
        '''
        display(HTML(html))
        print("✅ Auto-download triggered!")
    else:
        print("❌ Error: Could not generate JSON output.")
