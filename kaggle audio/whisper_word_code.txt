import os, time
from faster_whisper import WhisperModel
import glob

print("⏳ Loading Whisper large-v3 model...")
# Run on GPU with FP16
model_size = "large-v3"
model = WhisperModel(model_size, device="cuda", compute_type="float16")

# Find the latest generated audio
audio_dir = "/kaggle/working/outputs/v10_combine/"
wav_files = glob.glob(os.path.join(audio_dir, "*.wav"))
if not wav_files:
    print("❌ No audio files found in", audio_dir)
else:
    # Sort by modification time to get the latest
    latest_audio = max(wav_files, key=os.path.getctime)
    print(f"✅ Found audio: {latest_audio}")
    print("⏳ Transcribing with word-level timestamps... Please wait.")
    
    t0 = time.time()
    # word_timestamps=True allows word-by-word extraction
    segments, info = model.transcribe(latest_audio, beam_size=5, word_timestamps=True)
    
    print("Detected language '%s' with probability %f" % (info.language, info.language_probability))
    
    transcription_text = ""
    for segment in segments:
        for word in segment.words:
            # Saving with millisecond precision
            line = "[%.3fs -> %.3fs] %s" % (word.start, word.end, word.word)
            print(line)
            transcription_text += line + "\n"
            
    print(f"✅ Transcription finished in {time.time()-t0:.1f}s")
    
    # Write to text file
    txt_path = latest_audio.replace(".wav", "_word_transcription.txt")
    with open(txt_path, "w", encoding="utf-8") as f:
        f.write(transcription_text)
    
    print(f"📁 Word-level Transcription saved to: {txt_path}")
